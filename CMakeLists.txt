cmake_minimum_required(VERSION 3.18)
project(LeanStream VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
option(LEANSTREAM_PREDICTOR "Build the reference CPU predictor" ON)
if(LEANSTREAM_PREDICTOR)
    find_package(OpenMP REQUIRED)
endif()
set(RUNTIME_SOURCES src/runtime/runtime.cpp src/runtime/weight_manager.cpp
    src/runtime/execution_queue.cpp src/runtime/prediction_schedule.cpp
    src/runtime/kernel_selector.cpp src/io/stream_io.cpp)
add_library(leanstream_runtime STATIC ${RUNTIME_SOURCES})
target_include_directories(leanstream_runtime PUBLIC include)
target_link_libraries(leanstream_runtime PUBLIC Threads::Threads)
if(LEANSTREAM_PREDICTOR)
add_library(leanstream_predictor STATIC predictor/predictor_cpu.cpp)
target_include_directories(leanstream_predictor PUBLIC include)
target_link_libraries(leanstream_predictor PUBLIC OpenMP::OpenMP_CXX)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
    target_compile_options(leanstream_predictor PRIVATE -march=armv8.2-a+fp16+dotprod)
endif()
endif()
set(CPU_EXAMPLES dynamic_io prediction_schedule kernel_selection)
if(LEANSTREAM_PREDICTOR)
    list(APPEND CPU_EXAMPLES predictor)
endif()
foreach(example ${CPU_EXAMPLES})
    add_executable(${example}_example examples/${example}/main.cpp)
    target_include_directories(${example}_example PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(${example}_example PRIVATE leanstream_runtime)
    if(example STREQUAL "predictor")
        target_link_libraries(${example}_example PRIVATE leanstream_predictor)
    endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
    # Tests use assert for both checks and API calls, including Release builds.
    add_executable(runtime_test tests/open_source_runtime_test.cpp ${RUNTIME_SOURCES})
    target_include_directories(runtime_test PRIVATE include)
    target_compile_definitions(runtime_test PRIVATE STREAM_TESTING LEANSTREAM_RUNTIME_TESTING)
    target_link_libraries(runtime_test PRIVATE Threads::Threads)
    foreach(test kernel_selector prediction_schedule)
        add_executable(${test}_test tests/${test}_test.cpp)
        target_include_directories(${test}_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
        target_link_libraries(${test}_test PRIVATE leanstream_runtime)
    endforeach()
    foreach(test runtime kernel_selector prediction_schedule)
        target_compile_options(${test}_test PRIVATE -UNDEBUG)
        add_test(NAME ${test} COMMAND ${test}_test)
        set_tests_properties(${test} PROPERTIES TIMEOUT 30)
    endforeach()
    if(LEANSTREAM_PREDICTOR)
        add_test(NAME predictor_smoke COMMAND predictor_example)
    endif()
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
        add_test(NAME dynamic_io COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_io_example.py $<TARGET_FILE:dynamic_io_example> dynamic)
        add_test(NAME triton_imports COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_triton_imports.py)
        add_test(NAME selector_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_selector_contract.py)
    endif()
endif()
foreach(example ${CPU_EXAMPLES})
    target_compile_options(${example}_example PRIVATE -UNDEBUG)
endforeach()
option(LEANSTREAM_CUDA "Build optional CUDA examples" OFF)
if(LEANSTREAM_CUDA)
    enable_language(CUDA)
    set(CMAKE_CUDA_STANDARD 17)
    set(CMAKE_CUDA_STANDARD_REQUIRED ON)
    find_package(CUDAToolkit REQUIRED)
    add_library(leanstream_mlp STATIC kernels/streaming_mlp.cu kernels/streaming_mlp_candidates.cu)
    target_include_directories(leanstream_mlp PUBLIC include kernels)
    target_link_libraries(leanstream_mlp PUBLIC leanstream_runtime CUDA::cudart)
    set(CUDA_EXAMPLES persistent_consumer)
    if(LEANSTREAM_PREDICTOR)
        list(APPEND CUDA_EXAMPLES end_to_end)
    endif()
    foreach(example ${CUDA_EXAMPLES})
        add_executable(${example}_example examples/${example}/main.cu)
        target_include_directories(${example}_example PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
        target_compile_options(${example}_example PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-UNDEBUG>)
        target_link_libraries(${example}_example PRIVATE leanstream_mlp)
        if(LEANSTREAM_PREDICTOR)
            target_link_libraries(${example}_example PRIVATE leanstream_predictor)
        endif()
    endforeach()
    if(LEANSTREAM_PREDICTOR)
    add_executable(kernel_selection_cuda_example examples/kernel_selection/main.cu)
    target_include_directories(kernel_selection_cuda_example PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_options(kernel_selection_cuda_example PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-UNDEBUG>)
    target_link_libraries(kernel_selection_cuda_example PRIVATE leanstream_mlp leanstream_predictor)
    endif()
    if(BUILD_TESTING)
        add_test(NAME persistent_mlp COMMAND persistent_consumer_example)
        if(Python3_FOUND AND LEANSTREAM_PREDICTOR)
            add_test(NAME end_to_end COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_io_example.py $<TARGET_FILE:end_to_end_example> cuda)
            set_tests_properties(end_to_end PROPERTIES TIMEOUT 45 LABELS CUDA)
        endif()
        set_tests_properties(persistent_mlp PROPERTIES TIMEOUT 45 LABELS CUDA)
    endif()
endif()

option(LEANSTREAM_TRITON "Build native enqueue support for the production Triton backend" OFF)
if(LEANSTREAM_TRITON)
    find_package(CUDAToolkit REQUIRED)
    set_target_properties(leanstream_runtime PROPERTIES POSITION_INDEPENDENT_CODE ON)
    add_library(leanstream_triton SHARED backends/triton/enqueue.cpp backends/triton/mapped.cpp)
    target_include_directories(leanstream_triton PUBLIC include)
    target_link_libraries(leanstream_triton PUBLIC leanstream_runtime CUDA::cudart CUDA::cuda_driver)
endif()
